 *  │  Initial Mint   :   242,916.0 BTS  → distribution address          │

 *  │  Staking Lock   : 2,381,375.6 BTS  → held in contract              │

import {Ownable} from "@openzeppelin/contracts@5.0.2/access/Ownable.sol";

import {ERC20} from "@openzeppelin/contracts@5.0.2/token/ERC20/ERC20.sol";

import {ERC20Permit} from "@openzeppelin/contracts@5.0.2/token/ERC20/extensions/ERC20Permit.sol";


contract BravoTransStableCoin is ERC20, Ownable, ERC20Permit {

    // ─────────────────────────────────────────────────────────────────────────

    // Constants

    // ─────────────────────────────────────────────────────────────────────────


    /// @notice Total token supply: 2,624,291.6 BTS expressed in wei (18 decimals).

    uint256 public constant TOTAL_SUPPLY = 2_624_291_600_000_000_000_000_000;


    /// @notice Tokens minted immediately to the distribution address on deployment.

    uint256 public constant INITIAL_DISTRIBUTION = 242_916 * 10 ** 18;


    /// @notice Tokens locked inside the contract for staking and future allocations.

    uint256 public constant STAKING_RESERVE = TOTAL_SUPPLY - INITIAL_DISTRIBUTION;


    // ─────────────────────────────────────────────────────────────────────────

    // State Variables

    // ─────────────────────────────────────────────────────────────────────────


    bool public stakingSupplyReleased;

    address public immutable stakingVault;


    // ─────────────────────────────────────────────────────────────────────────

    // Events

    // ─────────────────────────────────────────────────────────────────────────


    event StakingSupplyReleased(address indexed to, uint256 amount);

    event TokensMinted(address indexed to, uint256 amount);


    // ─────────────────────────────────────────────────────────────────────────

    // Errors

    // ─────────────────────────────────────────────────────────────────────────


    error BTS__ZeroAddress();

    error BTS__StakingSupplyAlreadyReleased();

    error BTS__ZeroMintAmount();


    // ─────────────────────────────────────────────────────────────────────────

    // Constructor

    // ─────────────────────────────────────────────────────────────────────────


    constructor(

        address distributionAddress,

        address stakingVaultAddress,

        address initialOwner

    )

        ERC20("BravoTrans Stable Coin", "BTS")

        Ownable(initialOwner)

        ERC20Permit("BravoTrans Stable Coin")

    {

        if (distributionAddress == address(0)) revert BTS__ZeroAddress();

        if (stakingVaultAddress == address(0))  revert BTS__ZeroAddress();

        if (initialOwner == address(0))         revert BTS__ZeroAddress();


        stakingVault = stakingVaultAddress;


        _mint(distributionAddress, INITIAL_DISTRIBUTION);

        _mint(address(this), STAKING_RESERVE);

    }


    // ─────────────────────────────────────────────────────────────────────────

    // Owner-Only Functions

    // ─────────────────────────────────────────────────────────────────────────


    function mint(address to, uint256 amount) external onlyOwner {

        if (to == address(0))  revert BTS__ZeroAddress();

        if (amount == 0)       revert BTS__ZeroMintAmount();


        _mint(to, amount);

        emit TokensMinted(to, amount);

    }


    function releaseStakingSupply() external onlyOwner {

        if (stakingSupplyReleased) revert BTS__StakingSupplyAlreadyReleased();


        stakingSupplyReleased = true;

        uint256 amount = balanceOf(address(this));


        _transfer(address(this), stakingVault, amount);


        emit StakingSupplyReleased(stakingVault, amount);

    }


    // ─────────────────────────────────────────────────────────────────────────

    // View / Pure Helpers

    // ─────────────────────────────────────────────────────────────────────────


    function lockedStakingBalance() external view returns (uint256) {

        return balanceOf(address(this));

    }

}
